function processExcelData(rows) { let headerRowIndex = -1; for (let i = 0; i < rows.length; i++) { if (rows[i] && rows[i].some(cell => typeof cell === 'string' && cell.toLowerCase().includes('clinic name'))) { headerRowIndex = i; break; } } if (headerRowIndex === -1) { alert("Could not find 'Clinic Name' column in the uploaded file."); document.getElementById('loader').style.display = 'none'; return; } const headers = rows[headerRowIndex].map(h => typeof h === 'string' ? h.toLowerCase().trim() : ''); const clinicIdx = headers.findIndex(h => h.includes('clinic name')); const loginIdIdx = headers.findIndex(h => h.includes('login id')); const loginTimeIdx = headers.findIndex(h => h.includes('login time')); const logoutTimeIdx = headers.findIndex(h => h.includes('logout time')); const ipIdx = headers.findIndex(h => h.includes('ip')); const consultIdx = headers.findIndex(h => h.includes('consultation count')); const policyIdx = headers.findIndex(h => h.includes('policy count')); globalData = []; const regionsSet = new Set(); const codesSet = new Set(); const seenClinics = new Set(); // Unique Clinic track karne ke liye for (let i = headerRowIndex + 1; i < rows.length; i++) { const row = rows[i]; if (!row || !row[clinicIdx]) continue; const clinicName = (row[clinicIdx] || '').toString().trim(); if (!clinicName) continue; const loginId = loginIdIdx !== -1 ? (row[loginIdIdx] || '-').toString().trim() : '-'; // Clean Unique Key create karna (Clinic Name + Normalized Login ID basis par) const cleanId = loginId.replace(/^ec/i, '').trim(); const uniqueKey = `${clinicName}_${cleanId}`; // DUPLICATE REMOVAL: Agar ye clinic pehle aa chuka hai, toh skip kar do if (seenClinics.has(uniqueKey)) { continue; } seenClinics.add(uniqueKey); // Extract Login Code Prefix (e.g., "23:", "14:") let codePrefix = extractCodePrefix(loginId); if (codePrefix === 'OTHER') { codePrefix = extractCodePrefix(clinicName); } if (codePrefix !== 'OTHER') { codesSet.add(codePrefix); } // Region extraction (last word of clinic name) const parts = clinicName.split(' '); const region = parts.length > 1 ? parts[parts.length - 1].toUpperCase() : 'UNKNOWN'; regionsSet.add(region); globalData.push({ region: region, codePrefix: codePrefix, clinicName: clinicName, loginId: loginId, loginTime: loginTimeIdx !== -1 ? (row[loginTimeIdx] || '-').toString().trim() : '-', logoutTime: logoutTimeIdx !== -1 ? (row[logoutTimeIdx] || '-').toString().trim() : '-', ip: ipIdx !== -1 ? (row[ipIdx] || '-').toString().trim() : '-', consultCount: consultIdx !== -1 ? (row[consultIdx] || '0').toString().trim() : '0', policyCount: policyIdx !== -1 ? (row[policyIdx] || '0').toString().trim() : '0' }); } // Initialize Login Code Dropdown const codeSelect = document.getElementById('codeSelect'); codeSelect.innerHTML = ''; const sortedCodes = Array.from(codesSet).sort((a, b) => parseInt(a) - parseInt(b)); sortedCodes.forEach(code => { const opt = document.createElement('option'); opt.value = code; opt.textContent = code; codeSelect.appendChild(opt); }); codeSelect.disabled = false; // Initialize Region Dropdown const regionSelect = document.getElementById('regionSelect'); regionSelect.innerHTML = ''; const sortedRegions = Array.from(regionsSet).sort(); sortedRegions.forEach(reg => { const opt = document.createElement('option'); opt.value = reg; opt.textContent = reg; regionSelect.appendChild(opt); }); regionSelect.disabled = false; document.getElementById('loader').style.display = 'none'; // Initial state hidden rakhein ya filter apply karne par hi render karein renderTable(globalData); }
